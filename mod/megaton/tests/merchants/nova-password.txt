# Nova and the password to Moriarty's terminal (MG_F_HAS_PASSWORD = 64). She is
# only asked once the player knows of the terminal (MG_F_KNOWS_TERMINAL = 32).
# No leverage: refused. Then each way in turn: Charisma 8, Speech 50, a rented
# room, the back pay handed to Gob (MG_F_WAGES_RETURNED = 8388608).
#   python3 mod/megaton/tests/merchants/run_all.py nova-password
# The flags are put back with setgvar between the variants.

dude @NOVA/2/2
wait 800
# does not know of the terminal: nothing to ask
talk @NOVA
wait 2500
shot 01-hub-no-terminal
key 4
wait 1500

setgvar $GVAR_MG_FLAGS $MG_F_KNOWS_TERMINAL
talk @NOVA
wait 2500
shot 02-hub-plain-question
# 1 room, 2 price, 3 people, 4 the password, 5 see you
key 4
wait 1500
shot 03-refused
expect gvar $GVAR_MG_FLAGS $MG_F_KNOWS_TERMINAL
key 2
wait 1500
expect state dialog=0

# Charisma + 6 (rig command 9, stat 3)
setgvar 690 3
setgvar 691 6
setgvar 692 9
wait 500
talk @NOVA
wait 2500
shot 04-hub-charisma
key 4
wait 1500
shot 05-password
expect gvar $GVAR_MG_FLAGS 96
key 2
wait 1500

# has it: the question is gone
talk @NOVA
wait 2500
shot 06-hub-has-password
key 4
wait 1500
expect state dialog=0

# Speech + 60 (skill 14): the Speech line takes the place of the smile
setgvar $GVAR_MG_FLAGS $MG_F_KNOWS_TERMINAL
setgvar 690 14
setgvar 691 60
setgvar 692 10
wait 500
talk @NOVA
wait 2500
shot 07-hub-speech
key 4
wait 1500
expect gvar $GVAR_MG_FLAGS 96
key 2
wait 1500

# a paying guest
setgvar $GVAR_MG_FLAGS $MG_F_KNOWS_TERMINAL
talk @NOVA
wait 2500
key 1
wait 1500
key 1
wait 1500
# "[Go to bed.]": the night, and the player wakes in the rented room
key 1
wait 6500
expect state dialog=0
dude @NOVA/2/2
wait 800
talk @NOVA
wait 2500
shot 08-hub-regular
key 4
wait 1500
expect gvar $GVAR_MG_FLAGS 96
key 2
wait 1500

# after the back pay: her greeting changes and she simply tells
setgvar $GVAR_MG_FLAGS 8388640
talk @NOVA
wait 2500
shot 09-greeting-after-back-pay
key 4
wait 1500
shot 10-password-after-back-pay
expect gvar $GVAR_MG_FLAGS 8388704
key 2
wait 1500
expect state dialog=0

# she says her thanks once; after that it is the regular's greeting again
talk @NOVA
wait 2500
shot 11-greeting-later
key 4
wait 1500
expect state dialog=0
