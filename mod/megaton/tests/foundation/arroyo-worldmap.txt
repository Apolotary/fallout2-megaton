# What a player who started the normal way sees on the world map: leaving
# Arroyo by the bridge, Megaton is a circle south-east of the village that is
# not named yet (he has not been there).
#   python3 mod/megaton/test.py --group foundation --stage tests/foundation/stage.py --map arbridge.map --steps tests/foundation/arroyo-worldmap.txt
# 26734 is one of arbridge.map's exit-grid hexes that lead to the world map.

expect state map=ARBRIDGE.MAP
expect gvar $GVAR_MG_KIT_GIVEN $MG_KIT_UNDECIDED
# standing on an exit grid is enough: the world map opens by itself
dude 26734
wait 8000
shot 01-worldmap-from-arroyo
state

# walk to the unnamed circle (its centre is window 237,181) and step in: the
# player arrives on foot, so there is no starting kit for him
click 237 181
wait 12000
shot 02-on-the-circle
# the view has scrolled with the party: the marker is at window 195,141 now. (The
# TOWN button / T key do nothing here: they need a town-map picture, Megaton has none.)
click 195 141
wait 8000
expect state map=MEGATON.MAP mapIndex=151 dudeTile=@ENTRY
expect gvar $GVAR_MG_KIT_GIVEN $MG_KIT_NOT_DUE
shot 03-megaton-on-foot
key i
wait 1500
shot 04-no-kit
key esc
wait 800
# and back out: the circle has its name now
clicktile @GATE_EXIT
wait 8000
shot 05-named-after-visit
