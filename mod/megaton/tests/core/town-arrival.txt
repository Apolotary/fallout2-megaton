# REAL TOWN (no stage): the arrival sequence on the architect's map.
#   python3 mod/megaton/test.py --group core-town --only megaton,mgweld,mgsimms,mgburke,mgbomb,mgharden,mghitman,mgspgate,mgspsims,mgsprad,mggate,mghouse --steps tests/core/town-arrival.txt
# The gate is shut (frame 0) and locked by mggate.ssl; the first steps show the
# arrival line; Weld greets inside six hexes and the gate opens (frame 4): on
# the way to try the gate by hand the player already comes that close. The
# player walks through to @TRACK, inside the trigger around SIMMS_GREET; Simms
# comes over and starts the first-meeting conversation.
# Expected: flags WELD_GREETED | SIMMS_MET | SIMMS_SUMMONED (4456449), reputation +1.

expect state map=MEGATON.MAP mapIndex=151 elev=0 dudeTile=@ENTRY
expect obj @GATE 0 frame=0
shot 01-at-the-entrance
# trying the gate by hand: the walk up to it brings the player inside Weld's six hexes
use @GATE
wait 5000
shot 02-at-the-gate
gvar $GVAR_MG_FLAGS
# towards the robot
clicktile @WELD/1/3
wait 6000
shot 03-weld-greets
expect gvar $GVAR_MG_FLAGS 1
wait 3000
expect obj @GATE 0 frame=4
shot 04-gate-open
# through the gate
clicktile @GATE/5/2
wait 5000
center @GATE
clicktile @TRACK
wait 6000
center @TRACK
wait 400
shot 05-inside
wait 8000
expect state dialog=1
shot 06-simms-stops-you
key 1
wait 1500
key 5
wait 1500
expect state dialog=0
expect gvar $GVAR_MG_FLAGS 4456449
expect gvar $GVAR_TOWN_REP_MEGATON 1
# he walks back to his post
wait 12000
expect obj @SIMMS 0 name=Lucas Simms
center @SIMMS
wait 300
shot 07-simms-at-his-post
