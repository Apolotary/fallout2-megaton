# The town's grudge lapses: hostile, stamped on day 0, and the player walks in
# again eight days later.
#   python3 mod/megaton/tests/core/run_all.py amnesty
# The hostility is set by hand while the player is on the world map (on the
# compact stage a hostile town would be on him before he reached the exit).
# Expected after re-entry: TOWN_HOSTILE 0, HOSTILE_DAY 0, AMNESTY 1, reputation -15.

setgvar $GVAR_MG_FLAGS 262145
dude @RIG/0
wait 300
skill 12 @RIG
wait 6000
dude @RIG3/0
wait 300
use @RIG3
wait 6000
expect gvar 688 808
dude @GATE_EXIT/0/3
wait 300
clicktile @GATE_EXIT
wait 9000
setgvar $GVAR_MG_TOWN_HOSTILE $MG_TOWN_IS_HOSTILE
setgvar $GVAR_MG_HOSTILE_DAY 1
click 237 181
wait 9000
expect state mapIndex=151
shot 01-forgiven
expect gvar $GVAR_MG_TOWN_HOSTILE 0
expect gvar $GVAR_MG_HOSTILE_DAY 0
expect gvar $GVAR_MG_AMNESTY 1
expect gvar $GVAR_TOWN_REP_MEGATON -15
wait 4000
expect state combat=0
