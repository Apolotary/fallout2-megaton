# Walk the town in the real engine: through the gate, into buildings (roofs must vanish
# inside and come back outside), open and close doors, out by the exit grid.
#   python3 mod/megaton/test.py --group layout --only megaton --steps tests/layout/walk.txt --res 1280x960 --timeout 400
# --only megaton: doors carry no scripts, so the office and the armory open like any door
# (their locks belong to the side-quest and flavour groups and are tested there).
# @DOOR_* / @IN_* / @TRACK / @POOL / @PLAZA are written by layout.test_names().
expect state map=MEGATON.MAP mapIndex=151 elev=0 dudeTile=@ENTRY

# --- the gate (tests/layout/gate.txt looks at it closely)
use @GATE
wait 6000
expect obj @GATE 0 frame=4
shot 01-gate-open
clicktile @TRACK
wait 7000
expect state dudeTile=@TRACK
center @TRACK
wait 400
shot 02-inside-the-gate

# --- sheriff's house: front door, roof off, armory door, roof back on

expect state dudeTile=@IN_SHERIFF_ARMORY

expect state dudeTile=@TRACK
shot 05-out-again-roof-back
